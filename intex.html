    <script>

        // Counter Logic

        let riderCount = 1;

        const counter = document.getElementById('riderCount');

        const incrementBtn = document.getElementById('incrementBtn');

        const decrementBtn = document.getElementById('decrementBtn');

        const goBtn = document.getElementById('goBtn');

        const formSection = document.getElementById('formSection');

        const poolDashboard = document.getElementById('poolDashboard');

        const yourRides = document.getElementById('yourRides');

        const passengerList = document.getElementById('passengerList');


        incrementBtn.addEventListener('click', () => {

            riderCount = Math.min(riderCount + 1, 4);

            counter.textContent = riderCount;

        });


        decrementBtn.addEventListener('click', () => {

            riderCount = Math.max(riderCount - 1, 1);

            counter.textContent = riderCount;

        });


        goBtn.addEventListener('click', () => {

            document.querySelector('.counter-section').style.display = 'none';

            formSection.style.display = 'block';

        });


        // GPS Integration

        function getCurrentLocation() {

            if (navigator.geolocation) {

                navigator.geolocation.getCurrentPosition(position => {

                    const geocoder = new google.maps.Geocoder();

                    const latlng = {

                        lat: position.coords.latitude,

                        lng: position.coords.longitude

                    };


                    geocoder.geocode({

                        location: latlng

                    }, (results, status) => {

                        if (status === "OK" && results[0]) {

                            document.getElementById('pickup').value = results[0].formatted_address;

                        }

                    });

                }, error => {

                    alert('Unable to retrieve location. Please enter manually.');

                });

            } else {

                alert("Geolocation not supported by this browser.");

            }

        }


        // Pool Creation

        function createPool() {

            const pickup = document.getElementById('pickup').value;

            const drop = document.getElementById('drop').value;

            const time = document.getElementById('pickupTime').value;

            const cost = parseFloat(document.getElementById('cost').value);


            if (!pickup || !drop || !time || !cost) {

                alert('Please fill all fields');

                return;

            }


            // Store data in a pool object (or similar)

            this.poolData = {

                pickup,

                drop,

                time,

                cost,

                riders: 1

            };


            document.getElementById('displayRoute').innerText = `📍 ${pickup} → ${drop}`;

            document.getElementById('displayTime').innerText = `🕘 ${time} | 👥 ${this.poolData.riders}/4 riders`;

            document.getElementById('displayCost').innerText = `₹${cost}/person`;


            formSection.style.display = 'none';

            poolDashboard.style.display = 'block';

            createdPool.style.display = 'block';

        }


        // Show Join Form

        function showJoinForm() {

            joinForm.style.display = 'block';

            overlay.style.display = 'block';

        }


        // Show Passenger Details

        function showPassengerDetails() {

            passengerList.innerHTML = ''; // Clear previous details


            // Add the creator of the pool to passenger details

            const creatorDiv = document.createElement('div');

            creatorDiv.innerHTML = `<strong>Host:</strong> ${this.poolData.name} <br> <strong>Phone:</strong> ${this.poolData.phone}`;

            passengerList.appendChild(creatorDiv);


            // Iterate through the joinedPassengers array and display their details.


            if (this.joinedPassengers && Array.isArray(this.joinedPassengers)) {

                for (let i = 0; i < this.joinedPassengers.length; i++) {

                    const passenger = this.joinedPassengers[i];

                    const passengerDiv = document.createElement('div');

                    passengerDiv.innerHTML = `<strong>Name:</strong> ${passenger.name} <br> <strong>Phone:</strong> ${passenger.phone}`;

                    passengerList.appendChild(passengerDiv);

                }

            }


            passengerDetails.style.display = 'block';

            overlay.style.display = 'block';

        }


        // Hide Passenger Details

        function hidePassengerDetails() {

            passengerDetails.style.display = 'none';

            overlay.style.display = 'none';

        }


        // Pool Joining

        function joinPool() {

            const name = document.getElementById('name').value;

            const phone = document.getElementById('phone').value;


            if (!name || !phone) {

                alert('Please enter your name and phone number.');

                return;

            }


            if (this.poolData.riders < 4) {

                this.poolData.riders++;

                const splitCost = this.poolData.cost / this.poolData.riders;


                // Create a joinedPassengers Array

                if (!this.joinedPassengers) {

                    this.joinedPassengers = [];

                }


                // Add user to that array

                this.joinedPassengers.push({

                    name,

                    phone

                });


                // Store the person creating the pool

                this.poolData.name = name;

                this.poolData.phone = phone;


                // Update the available pool details

                document.getElementById('displayTime').innerText = `🕘 ${this.poolData.time} | 👥 ${this.poolData.riders}/4 riders`;

                document.getElementById('displayCost').innerText = `₹${splitCost.toFixed(2)}/person`;


                // Display joined pool details in "Your Rides"

                document.getElementById('joinedDisplayRoute').innerText = `📍 ${this.poolData.pickup} → ${this.poolData.drop}`;

                document.getElementById('joinedDisplayTime').innerText = `🕘 ${this.poolData.time} | 👥 ${this.poolData.riders}/4 riders`;

                document.getElementById('joinedDisplayCost').innerText = `₹${splitCost.toFixed(2)}/person`;


                yourRides.style.display = 'block'; // Show "Your Rides" section


                // Hide the join form and overlay

                joinForm.style.display = 'none';

                overlay.style.display = 'none';

            } else {

                alert('Pool is full!');

            }

        }
